import { Router, Request, Response } from 'express';
import { PrismaClient } from '@prisma/client';
import { tenantIdentification } from './tenantMiddleware.js';

const router = Router();
const prisma = new PrismaClient();

// Apply the tenant validation middleware to all stream management endpoints
router.use(tenantIdentification);

/**
 * GET /api/streams/active
 * Fetches the current live stream data for the validated streamer workspace
 */
router.get('/active', async (req: Request, res: Response) => {
  try {
    const stream = await prisma.stream.findFirst({
      where: {
        tenantId: req.tenantId,
        status: 'LIVE'
      },
      select: {
        id: true,
        title: true,
        description: true,
        status: true,
        viewerCount: true
      }
    });

    if (!stream) {
      return res.json({ 
        status: 'OFFLINE', 
        message: 'This workspace is currently not broadcasting.' 
      });
    }

    res.json(stream);
  } catch (error) {
    res.status(500).json({ 
      error: 'Query Failure', 
      message: 'Could not retrieve active stream configurations.' 
    });
  }
});

/**
 * POST /api/streams/viewer-heartbeat
 * Increments or updates current active viewer counts securely for the tenant
 */
router.post('/viewer-heartbeat', async (req: Request, res: Response) => {
  const { streamId, action } = req.body; // action can be 'join' or 'leave'

  if (!streamId) {
    return res.status(400).json({ error: 'Missing streamId parameter.' });
  }

  try {
    const changeAmount = action === 'leave' ? -1 : 1;

    const updatedStream = await prisma.stream.update({
      where: { 
        id: streamId,
        tenantId: req.tenantId // Secondary check ensures strict cross-tenant isolation
      },
      data: {
        viewerCount: {
          increment: changeAmount
        }
      },
      select: { viewerCount: true }
    });

    // Prevent negative counts from bad syncs
    if (updatedStream.viewerCount < 0) {
      await prisma.stream.update({
        where: { id: streamId },
        data: { viewerCount: 0 }
      });
      updatedStream.viewerCount = 0;
    }

    res.json({ success: true, activeViewers: updatedStream.viewerCount });
  } catch (error) {
    res.status(500).json({ 
      error: 'Update Failure', 
      message: 'Failed to synchronize live audience counts.' 
    });
  }
});

export default router;
